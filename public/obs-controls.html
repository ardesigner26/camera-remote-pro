// ============================================ // ============================================ // SISTEMA ANTI-TELA-PRETA // ============================================ function setupAutoReconnect() { if (!peerConnection) { console.log('⚠️ peerConnection não existe ainda, tentando novamente...'); setTimeout(setupAutoReconnect, 1000); return; } console.log('🔧 Configurando reconexão automática...'); peerConnection.addEventListener('connectionstatechange', () => { console.log('🔌 Estado da conexão:', peerConnection.connectionState); if (peerConnection.connectionState === 'failed' || peerConnection.connectionState === 'disconnected') { console.warn('⚠️ Conexão perdida! Reconectando em 2 segundos...'); setTimeout(() => { console.log('🔄 Forçando reload para reconectar...'); location.reload(); }, 2000); } }); peerConnection.addEventListener('iceconnectionstatechange', () => { console.log('❄️ ICE estado:', peerConnection.iceConnectionState); if (peerConnection.iceConnectionState === 'failed' || peerConnection.iceConnectionState === 'disconnected') { console.warn('⚠️ ICE connection falhou! Reconectando...'); setTimeout(() => { location.reload(); }, 2000); } }); console.log('✅ Auto-reconnect configurado!'); } let wasHidden = false; document.addEventListener('visibilitychange', () => { if (document.hidden) { console.log('👁️ Browser Source OCULTO (cena mudou)'); wasHidden = true; } else { console.log('👁️ Browser Source VISÍVEL novamente'); if (wasHidden) { setTimeout(() => { if (peerConnection) { const state = peerConnection.connectionState; console.log('🔍 Verificando conexão após voltar:', state); if (state === 'failed' || state === 'disconnected' || state === 'closed') { console.warn('❌ Conexão morreu durante mudança de cena!'); console.log('🔄 Recarregando página...'); location.reload(); } } else { console.warn('❌ peerConnection não existe!'); location.reload(); } }, 1000); wasHidden = false; } } }); setInterval(() => { if (peerConnection && peerConnection.connectionState === 'connected') { peerConnection.getStats().then(stats => { stats.forEach(report => { if (report.type === 'inbound-rtp' && report.kind === 'video') { const bytesReceived = report.bytesReceived || 0; if (window.lastBytesReceived === undefined) { window.lastBytesReceived = bytesReceived; window.noDataCounter = 0; } else { if (bytesReceived === window.lastBytesReceived) { window.noDataCounter++; console.warn(⚠️ Sem dados novos há + (window.noDataCounter * 5) + 's'); if (window.noDataCounter >= 3) { console.error('❌ 15s sem dados! Reconectando...'); location.reload(); } } else { window.noDataCounter = 0; window.lastBytesReceived = bytesReceived; } } } }); }); } }, 5000); setTimeout(() => { console.log('🚀 Iniciando sistema anti-tela-preta...'); setupAutoReconnect(); }, 3000);
