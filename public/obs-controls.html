}); return; } peerConnection.getStats(null).then(stats => { let resolution = '--'; let fps = '--'; let bitrate = '--'; let codec = '--'; let packetLoss = '--'; stats.forEach(report => { // Capturar stats de vídeo recebido if (report.type === 'inbound-rtp' && report.kind === 'video') { // Resolução if (report.frameWidth && report.frameHeight) { resolution = report.frameWidth + 'x' + report.frameHeight; } // FPS if (report.framesPerSecond) { fps = report.framesPerSecond.toFixed(1); } // Bitrate (calcular) if (report.bytesReceived !== undefined) { if (window.lastBytesReceived !== undefined && window.lastStatsTime !== undefined) { const bytesDiff = report.bytesReceived - window.lastBytesReceived; const timeDiff = (report.timestamp - window.lastStatsTime) / 1000; const bitrateKbps = Math.round((bytesDiff * 8) / timeDiff / 1000); bitrate = bitrateKbps + ' kbps'; } window.lastBytesReceived = report.bytesReceived; window.lastStatsTime = report.timestamp; } // Packet Loss if (report.packetsLost !== undefined && report.packetsReceived !== undefined) { const totalPackets = report.packetsLost + report.packetsReceived; if (totalPackets > 0) { const lossPercent = (report.packetsLost / totalPackets) * 100; packetLoss = lossPercent.toFixed(2) + '%'; } else { packetLoss = '0.00%'; } } // Codec if (report.codecId) { stats.forEach(codecReport => { if (codecReport.id === report.codecId && codecReport.mimeType) { codec = codecReport.mimeType; } }); } } }); // Atualizar elementos da interface const resEl = document.querySelector('[data-stat="resolution"]'); const fpsEl = document.querySelector('[data-stat="fps"]'); const bitrateEl = document.querySelector('[data-stat="bitrate"]'); const codecEl = document.querySelector('[data-stat="codec"]'); const lossEl = document.querySelector('[data-stat="loss"]'); const statusEl = document.querySelector('[data-stat="status"]'); if (resEl) resEl.textContent = resolution; if (fpsEl) fpsEl.textContent = fps; if (bitrateEl) bitrateEl.textContent = bitrate; if (codecEl) codecEl.textContent = codec; if (lossEl) lossEl.textContent = packetLoss; if (statusEl) statusEl.textContent = 'Conectado'; console.log('📊 Stats:', { resolution, fps, bitrate, codec, packetLoss }); }).catch(err => { console.error('❌ Erro ao obter stats:', err); }); } // Atualizar stats a cada 1 segundo setInterval(updateStatsDisplay, 1000);
